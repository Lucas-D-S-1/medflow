-- UPDATE: Atualizar estado civil do usuário
UPDATE cidade_alfa_mobilidade.bronze.tb_mob_usuario
SET estado_civil = 2
WHERE id_pessoa = 103;

/*
Objetivo: Simular a atualização de um dado cadastral do usuário no sistema.
O campo estado_civil é alterado de 1 para 2, representando uma mudança
no perfil do cidadão. No projeto real, este UPDATE seria acionado quando
o usuário atualiza seus dados no aplicativo de mobilidade da Cidade Alfa.
*/


-- SELECT: Confirmar UPDATE
SELECT * FROM cidade_alfa_mobilidade.bronze.tb_mob_usuario
WHERE id_pessoa = 103;

/*
Objetivo: Verificar se a atualização foi aplicada corretamente. O retorno
deve exibir o campo estado_civil com valor 2, confirmando que o UPDATE
foi executado com sucesso pela engine Delta Lake do Databricks.
*/


-- DELETE: Remover usuário de teste
DELETE FROM cidade_alfa_mobilidade.bronze.tb_mob_usuario

Objetivo: Simular a remoção de um registro do sistema, representando

-- SELECT: Confirmar DELETE

Objetivo: Verificar se o registro foi removido corretamente. O retorno

-- CREATE TABLE: Criar nova tabela processada na camada Silver
CREATE TABLE cidade_alfa_mobilidade.silver.tb_mob_usario_female_users
AS SELECT * FROM cidade_alfa_mobilidade.bronze.tb_mob_usario
WHERE sexo = 'F';

/*
Objetivo: Demonstrar o fluxo Bronze -> Silver da arquitetura Medallion.
A partir dos dados brutos da camada Bronze, é criada uma tabela processada
na camada Silver contendo apenas os registros de usuárias do sexo feminino.
Esta operação representa uma transformação e movimentação de dados entre
camadas, elevando a maturidade dos dados de bruto para estruturado e filtrado.
No projeto de mobilidade da Cidade Alfa, a camada Silver é onde vivem os dados
limpos e prontos para análise, como as tabelas T_MOB_ definidas no MER físico.
*/


-- SELECT: Confirmar nova tabela na Silver
SELECT * FROM cidade_alfa_mobilidade.silver.tb_mob_usario_female_users;

/*
Objetivo: Validar que a tabela foi criada corretamente na camada Silver
com os dados filtrados. O retorno deve exibir apenas registros com
sexo = 'F', confirmando que a transformação Bronze -> Silver foi
executada com sucesso.
*/
